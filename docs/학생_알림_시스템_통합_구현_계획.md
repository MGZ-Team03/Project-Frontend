### 📊 WebSocket 아키텍처

#### 두 개의 서로 다른 WebSocket 훅 존재

| 파일 | 용도 | 특징 |
|------|------|------|
| `/hooks/useWebSocket.js` | 알림 수신 전용 | ✅ onMessage 콜백 지원 |
| `/hooks/webSocket/useWebSocket.js` | 상태 전송 전용 | ❌ onMessage 지원 안 함 |

#### 현재 사용 현황

| 컴포넌트 | 사용하는 훅 | 기능 |
|----------|------------|------|
| TutorSearchDialog.jsx | `/hooks/useWebSocket.js` | 알림 수신 (승인/거부) |
| HomePage.jsx | `/hooks/webSocket/useWebSocket.js` | 상태 전송 (5초 간격) |

### ❌ 문제점

| 상황 | HomePage WebSocket | TutorSearchDialog WebSocket | 결과 |
|------|-------------------|---------------------------|------|
| HomePage만 열림 | ✅ 연결됨 (핸들러 없음) | ❌ 연결 안 됨 | ❌ 알림 못 받음 |
| TutorSearchDialog 열림 | ✅ 연결됨 (핸들러 없음) | ✅ 연결됨 (핸들러 있음) | ✅ 스낵바 뜸 |

### 🔧 해결 방법: webSocketConfig.js 수정

#### 3.1 webSocketConfig.js에 메시지 리스너 시스템 추가

**파일:** `/src/config/webSocketConfig.js`

```javascript
import {WS_URL} from "../utils/constants.js";

class WebSocketSingleton {
    constructor() {
        this.socket = null;
        this.intervalId = null;
        this.messageListeners = new Set();  // ← 추가: 메시지 리스너 집합
    }

    connect() {
        if (!this.socket || this.socket.readyState === WebSocket.CLOSED) {
            this.socket = new WebSocket(WS_URL);
            this._setupListeners();
        }
        return this.socket;
    }

    // ✅ 추가: 메시지 리스너 등록
    addMessageListener(listener) {
        this.messageListeners.add(listener);
        return () => this.messageListeners.delete(listener);  // unsubscribe 함수 반환
    }

    // ✅ 추가: 메시지 리스너 제거
    removeMessageListener(listener) {
        this.messageListeners.delete(listener);
    }

    startSendingData(interval = 50000, getData = () => null) {
        if (this.intervalId) return;

        this.intervalId = setInterval(() => {
            if (this.socket && this.socket.readyState === WebSocket.OPEN) {
                const data = getData();
                if (!data) {
                    console.log("❌ data null");
                    return;
                }
                this.socket.send(JSON.stringify(data));
                console.log("📤 데이터 전송:", data);
            }
        }, interval);
    }

    stopSendingData() {
        if (this.intervalId) {
            clearInterval(this.intervalId);
            this.intervalId = null;
        }
    }

    disconnect() {
        this.stopSendingData();
        if (this.socket) {
            this.socket.close();
            this.socket = null;
        }
    }

    _setupListeners() {
        this.socket.onopen = () => console.log("✅ WebSocket 연결됨");
        
        // ✅ 수정: 모든 등록된 리스너에게 메시지 전달
        this.socket.onmessage = (event) => {
            console.log("📩 메시지 수신:", event.data);
            
            try {
                const data = JSON.parse(event.data);
                
                // 모든 리스너에게 메시지 전달
                this.messageListeners.forEach(listener => {
                    try {
                        listener(data);
                    } catch (err) {
                        console.error("❌ 리스너 오류:", err);
                    }
                });
            } catch (err) {
                console.error("❌ 메시지 파싱 오류:", err);
            }
        };
        
        this.socket.onerror = (error) => console.error("❌ WebSocket 에러:", error);
        this.socket.onclose = () => console.log("⚡ WebSocket 연결 종료");
    }

    getSocket() {
        return this.socket;
    }
}

const ws = new WebSocketSingleton();
export default ws;
```

#### 3.2 useWebSocket.js (상태 전송용) 수정

**파일:** `/src/hooks/webSocket/useWebSocket.js`

```javascript
import { useEffect, useRef } from "react";
import ws from "../../config/webSocketConfig.js";

export default function useWebSocket(getData, options = {}) {
    const {
        interval = 5000,
        sendImmediately = true,
        enableInterval = true,
        onMessage = null  // ← 추가: 메시지 핸들러 옵션
    } = options;

    const getDataRef = useRef(getData);
    const onMessageRef = useRef(onMessage);

    useEffect(() => {
        getDataRef.current = getData;
    }, [getData]);

    useEffect(() => {
        onMessageRef.current = onMessage;
    }, [onMessage]);

    useEffect(() => {
        console.log("🔌 WebSocket 마운트");

        const socket = ws.connect();

        // ✅ 추가: 메시지 리스너 등록
        let unsubscribe = null;
        if (onMessageRef.current) {
            unsubscribe = ws.addMessageListener((data) => {
                if (onMessageRef.current) {
                    onMessageRef.current(data);
                }
            });
        }

        const handleOpen = () => {
            console.log("✅ WebSocket 열림");

            if (sendImmediately) {
                const data = getDataRef.current();
                if (data) {
                    socket.send(JSON.stringify(data));
                    console.log("📤 초기 전송:", data);
                } else {
                    console.log("⏭️ 초기 전송 생략 (getData가 null 반환)");
                }
            }

            if (enableInterval) {
                ws.startSendingData(interval, () => getDataRef.current());
            }
        };

        if (socket.readyState === WebSocket.OPEN) {
            handleOpen();
        } else {
            socket.addEventListener('open', handleOpen);
        }

        return () => {
            console.log("👋 WebSocket cleanup");
            socket.removeEventListener('open', handleOpen);
            if (enableInterval) {
                ws.stopSendingData();
            }
            // ✅ 추가: 메시지 리스너 제거
            if (unsubscribe) {
                unsubscribe();
            }
        };
    }, [interval, sendImmediately, enableInterval]);

    return ws.getSocket();
}
```
